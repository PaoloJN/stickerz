# Stack Stickers — Phase E1: Sticker Editor on svgl — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A full-screen, Framer-style `/editor` inside the svgl fork. Users place svgl logos as stickers on a true-scale MacBook lid, then style, arrange, share (URL) and export (PNG) the result.

**Architecture:**
- Pure TypeScript model in `src/lib/stickers/` (types, laptop specs, logo adapter over svgl data, geometry, history, codec, storage), unit-tested with Vitest.
- A Svelte 5 runes `EditorState` class drives the Svelte components in `src/components/editor/`.
- Logos are rendered as SVG `<image>` elements, never inlined.
- svgl files are touched minimally: root layout bypass, a sidebar link, and a card button.

**Tech Stack:** SvelteKit 2, Svelte 5 runes, Tailwind v4, shadcn-svelte (bits-ui), @lucide/svelte, fuse.js (already present), lz-string (new), Vitest (new).

**Spec:** `docs/superpowers/specs/2026-10-04-editor-on-svgl-design.md`

## Global Constraints

- Keep svgl's existing behaviour intact. Diffs to existing svgl files are limited to: `src/routes/+layout.svelte`, the sidebar links component, `src/components/svgs/svgCard.svelte`, `package.json`, `vite.config.ts`.
- New deps: `lz-string` (dependency), `vitest` (devDependency). Nothing else.
- All geometry is in millimetres. Lid sizes (mm): air13 304.1×215.0, air15 340.4×237.6, pro14 312.6×221.2, pro16 355.7×248.1. No Apple logo.
- Logo art is drawn only via SVG `<image href>`. Never inline logo SVG markup into the page DOM.
- localStorage key `ss:layout`; every storage access is wrapped in try/catch.
- Share URL format: `/editor#l=<lz-string encoded compact layout>`. The add-logo entry point is `/editor?add=<logoKey>`.
- Imports use svgl's `@/` alias (maps to `src/`).
- Style: svgl tokens + shadcn-svelte components. Floating panels use `rounded-xl border border-neutral-200 dark:border-neutral-800 bg-white/90 dark:bg-neutral-950/90 backdrop-blur shadow-lg`. Neutral palette only.
- Respect `prefers-reduced-motion` (no springs; instant moves).
- Dev server in the agent sandbox: `CHOKIDAR_USEPOLLING=1 pnpm dev`.
- Every commit message ends with a blank line, then `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

1. **Garbage or hostile share hashes** (`#l=xyz`, truncated, huge arrays, NaN coordinates, unknown logos) → the editor loads with an empty or partial valid layout and never throws. Pinned in Task 2 codec tests.
2. **localStorage blocked or corrupt** → the editor works without persistence. Pinned in Task 2 storage tests.
3. **Logo key collisions or svgl dataset updates** → keys stay unique and unknown keys are dropped. Pinned in Task 2 logos tests.
4. **Rotated-sticker resize/rotate math** (pointer on any side, any angle) → width follows the pointer distance in the sticker's local space and is clamped to 10–200 mm. Pinned in Task 2 geometry tests.
5. **Undo after many edits; continuous drags** → one history entry per gesture (pointerdown→pointerup), not per pointermove. Pinned in Task 4 by the `beginGesture`/`endGesture` contract and its test in Task 3.

```ts
import type { iSVG, ThemeOptions } from "@/types/svg";
import { svgs } from "@/data/svgs";

export type StickerLogo = {
  key: string;
  title: string;
  categories: string[];
  /** light = art for light backgrounds (white sticker), dark = art for dark backgrounds */
  icon: { light: string; dark?: string };
  wordmark?: { light: string; dark?: string };
};

export function logoKey(svg: { route: string | ThemeOptions }): string {
  const route = typeof svg.route === "string" ? svg.route : svg.route.light;
  const file = route.split("/").pop() ?? route;
  return file.replace(/\.svg$/i, "").replace(/[-_](light|dark)$/i, "");
}

const themed = (r: string | ThemeOptions) => (typeof r === "string" ? { light: r } : { light: r.light, dark: r.dark });

export function buildLogos(list: iSVG[]): StickerLogo[] {
  const seen = new Set<string>();
  const out: StickerLogo[] = [];
  for (const svg of list) {
    const key = logoKey(svg);
    if (seen.has(key)) continue;
    seen.add(key);
    out.push({
      key,
      title: svg.title,
      categories: Array.isArray(svg.category) ? svg.category : [svg.category],
      icon: themed(svg.route),
      ...(svg.wordmark ? { wordmark: themed(svg.wordmark) } : {}),
    });
  }
  return out;
}

export const LOGOS: StickerLogo[] = buildLogos(svgs);
export const LOGO_BY_KEY = new Map(LOGOS.map((l) => [l.key, l]));
export const isKnownLogo = (key: string) => LOGO_BY_KEY.has(key);

export function artFor(logo: StickerLogo, variant: "icon" | "wordmark", theme: "light" | "dark"): string {
  const art = variant === "wordmark" && logo.wordmark ? logo.wordmark : logo.icon;
  return theme === "dark" ? (art.dark ?? art.light) : art.light;
}
```

```ts
import type { StickerShape } from "./types";

export const MIN_W = 10;
export const MAX_W = 200;
const DEFAULT_W = 50;

export const clamp = (n: number, lo: number, hi: number) => Math.min(hi, Math.max(lo, n));
export const clampWidth = (w: number) => (Number.isFinite(w) ? clamp(w, MIN_W, MAX_W) : DEFAULT_W);

export function normalizeDeg(d: number): number {
  const r = ((d % 360) + 360) % 360; // [0,360)
  return r > 180 ? r - 360 : r === 0 ? 0 : r === 180 ? 180 : r;
}

export function rotatePoint(px: number, py: number, cx: number, cy: number, deg: number) {
  const a = (deg * Math.PI) / 180;
  const dx = px - cx;
  const dy = py - cy;
  return { x: cx + dx * Math.cos(a) - dy * Math.sin(a), y: cy + dx * Math.sin(a) + dy * Math.cos(a) };
}

/** Angle for a rotate handle that sits above the sticker (0 = handle straight up). */
export function rotationFromPointer(cx: number, cy: number, px: number, py: number, snap = false): number {
  let deg = (Math.atan2(py - cy, px - cx) * 180) / Math.PI + 90;
  if (snap) deg = Math.round(deg / 15) * 15;
  return normalizeDeg(deg);
}

/** Symmetric resize about the centre: width from the pointer's distance in the sticker's local space. */
export function widthFromPointer(s: { x: number; y: number; rot: number }, px: number, py: number, hOverW: number): number {
  const local = rotatePoint(px, py, s.x, s.y, -s.rot);
  const fromW = Math.abs(local.x - s.x) * 2;
  const fromH = hOverW > 0 ? (Math.abs(local.y - s.y) * 2) / hOverW : 0;
  return clampWidth(Math.max(fromW, fromH));
}

/** aspect = art width / height. Non-diecut shapes are square. */
export function stickerBox(s: { w: number; shape: StickerShape }, aspect: number): { w: number; h: number } {
  if (s.shape !== "diecut" || !(aspect > 0)) return { w: s.w, h: s.w };
  return { w: s.w, h: s.w / aspect };
}
```
Note: `normalizeDeg(-180)` must return `180` and `normalizeDeg(720)` must return `0`. The implementation above handles both. Keep the tests as the source of truth.

```ts
export class History<T> {
  private past: T[] = [];
  private future: T[] = [];
  constructor(private limit = 100) {}

  push(state: T) {
    this.past.push(state);
    if (this.past.length > this.limit) this.past.shift();
    this.future = [];
  }
  undo(current: T): T | null {
    const prev = this.past.pop();
    if (prev === undefined) return null;
    this.future.push(current);
    return prev;
  }
  redo(current: T): T | null {
    const next = this.future.pop();
    if (next === undefined) return null;
    this.past.push(current);
    return next;
  }
  get canUndo() { return this.past.length > 0; }
  get canRedo() { return this.future.length > 0; }
  clear() { this.past = []; this.future = []; }
}
```

```ts
import { compressToEncodedURIComponent, decompressFromEncodedURIComponent } from "lz-string";
import { clamp, clampWidth, normalizeDeg } from "./geometry";
import { newId } from "./id";
import { DEFAULT_LAPTOP, LAPTOPS } from "./laptops";
import { isKnownLogo } from "./logos";
import { COLORS, MODELS, SHAPES, type Layout, type Sticker } from "./types";

export const MAX_STICKERS = 200;
const isObj = (v: unknown): v is Record<string, unknown> => typeof v === "object" && v !== null && !Array.isArray(v);
const oneOf = <T extends string>(v: unknown, list: readonly T[], d: T): T => (list.includes(v as T) ? (v as T) : d);
const num = (v: unknown, d: number) => (typeof v === "number" && Number.isFinite(v) ? v : d);

export function sanitizeLayout(input: unknown, known: (k: string) => boolean = isKnownLogo): Layout | null {
  if (!isObj(input) || input.v !== 1) return null;
  const lap = isObj(input.laptop) ? input.laptop : {};
  const model = oneOf(lap.model, MODELS, DEFAULT_LAPTOP.model);
  const color = oneOf(lap.color, COLORS, DEFAULT_LAPTOP.color);
  const lid = LAPTOPS[model];
  const raw = Array.isArray(input.stickers) ? input.stickers : [];
  const stickers: Sticker[] = [];
  for (const s of raw) {
    if (stickers.length >= MAX_STICKERS) break;
    if (!isObj(s) || typeof s.logo !== "string" || !known(s.logo)) continue;
    stickers.push({
      id: typeof s.id === "string" && s.id ? s.id : newId(),
      logo: s.logo,
      variant: oneOf(s.variant, ["icon", "wordmark"] as const, "icon"),
      theme: oneOf(s.theme, ["light", "dark"] as const, "light"),
      shape: oneOf(s.shape, SHAPES, "diecut"),
      holo: s.holo === true,
      x: clamp(num(s.x, lid.w / 2), 0, lid.w),
      y: clamp(num(s.y, lid.h / 2), 0, lid.h),
      w: clampWidth(num(s.w, Number.NaN)),
      rot: normalizeDeg(num(s.rot, 0)),
    });
  }
  return { v: 1, laptop: { model, color }, stickers };
}

const r1 = (n: number) => Math.round(n * 10) / 10;

export function encodeLayout(l: Layout): string {
  const compact = [
    1, l.laptop.model, l.laptop.color,
    l.stickers.map((s) => [s.logo, s.variant === "wordmark" ? 1 : 0, s.theme === "dark" ? 1 : 0, SHAPES.indexOf(s.shape), s.holo ? 1 : 0, r1(s.x), r1(s.y), r1(s.w), Math.round(s.rot)]),
  ];
  return compressToEncodedURIComponent(JSON.stringify(compact));
}

export function decodeLayout(str: string, known: (k: string) => boolean = isKnownLogo): Layout | null {
  try {
    const json = decompressFromEncodedURIComponent(str);
    if (!json) return null;
    const c: unknown = JSON.parse(json);
    if (!Array.isArray(c) || c[0] !== 1 || !Array.isArray(c[3])) return null;
    return sanitizeLayout({
      v: 1,
      laptop: { model: c[1], color: c[2] },
      stickers: (c[3] as unknown[]).slice(0, MAX_STICKERS).map((t) => Array.isArray(t)
        ? { logo: t[0], variant: t[1] === 1 ? "wordmark" : "icon", theme: t[2] === 1 ? "dark" : "light", shape: SHAPES[t[3] as number], holo: t[4] === 1, x: t[5], y: t[6], w: t[7], rot: t[8] }
        : null),
    }, known);
  } catch {
    return null;
  }
}
```

```ts
export const LAYOUT_KEY = "ss:layout";

export function loadLayout(storage: Pick<Storage, "getItem"> | null, known: (k: string) => boolean = isKnownLogo): Layout | null {
  try {
    const raw = storage?.getItem(LAYOUT_KEY);
    return raw ? sanitizeLayout(JSON.parse(raw), known) : null;

    return null;

}

export function saveLayout(storage: Pick<Storage, "setItem"> | null, layout: Layout): void {
  try {
    storage?.setItem(LAYOUT_KEY, JSON.stringify(layout));
  } catch {
    // storage unavailable: editor keeps working without persistence
  }
}
```
